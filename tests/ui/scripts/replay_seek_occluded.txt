# replay scrubbing goes through the ui hit test: a press on the playfield seeks (hud_seekarea holds it), a press on
# the options overlay in front of it does not. watches the instant replay (F2 from the pause menu) of the playable
# fixture, as in mods_replay_speed. assertion-only; the first 'exit' only stops the replay.
options_save_on_back 0
debug_fixed_frametime 0.0166667
ui_scale_to_dpi 0
mouse_raw_input 0
mouse_sensitivity 1
windowed_resolution 1280x720
mod_nofail 1
mouse_to 10 10
@wait 30
osu_folder uitest_osu_folder_play
set_active_ui_screen songbrowser
@wait 300
sendkey Return
@wait_secs 5
ui_assert visible songbrowser 0
@sleep 4000
@wait 12
sendkey Escape
@wait 12
ui_assert visible pauseoverlay 1
sendkey F2
@wait 30
ui_assert visible pauseoverlay 0
mouse_to 900 360
@wait 2
mouse_down
@wait 3
ui_assert active hud_seekarea 1
mouse_to 950 360
@wait 3
ui_assert active hud_seekarea 1
mouse_up
@wait 3
ui_assert active hud_seekarea 0
sendkey Left Ctrl down
sendkey O
sendkey Left Ctrl up
@wait 40
ui_assert visible optionsoverlay 1
mouse_to 150 360
@wait 2
mouse_down
@wait 3
ui_assert active hud_seekarea 0
mouse_up
@wait 3
exit
@sleep 1000
@wait 60
mod_nofail 0
exit
