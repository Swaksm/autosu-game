# retrying from the pause menu must start the map with no hitobject done (they'd be invisible until their time).
# gameplay key releases are queued even while paused, stamped with the paused music position; one in the same frame
# as the retry click used to survive the restart and get judged on the next frame, at that position, missing every
# object up to it. (the Z release goes in the same frame as the mouse_up that clicks retry)
# no debug_fixed_frametime: the music runs on wall-clock time, so the waits are in seconds.
options_save_on_back 0
ui_scale_to_dpi 0
windowed_resolution 1280x720
mod_nofail 1
mouse_to 10 10
@wait 30
osu_folder uitest_osu_folder_play
set_active_ui_screen songbrowser
@wait 300
sendkey Return
@wait_secs 10
ui_assert visible songbrowser 0
sendkey Escape
@wait_secs 1
ui_assert visible pauseoverlay 1
mouse_to 640 360
@wait 5
mouse_down
@wait 2
sendkey Z up
mouse_up
@wait_secs 0.5
ui_assert visible pauseoverlay 0
ui_assert play finished 0
exit
